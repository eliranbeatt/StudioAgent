import { mutation, query } from "./_generated/server";
import { v } from "convex/values";
import { normalizeName, newBusinessId } from "./lib/normalize";

export const findOrCreateByName = mutation({
  args: { name: v.string() },
  handler: async (ctx, { name }) => {
    const trimmed = name.trim();
    if (!trimmed) throw new Error("Customer name is required");
    const n = normalizeName(trimmed);

    const existing = await ctx.db
      .query("customers")
      .withIndex("by_nameNormalized", (q) => q.eq("nameNormalized", n))
      .first();

    if (existing) return existing._id;

    // ensure unique business id
    for (let i = 0; i < 5; i++) {
      const cid = newBusinessId("CUST");
      const clash = await ctx.db
        .query("customers")
        .withIndex("by_customerId", (q) => q.eq("customerId", cid))
        .first();
      if (!clash) {
        return await ctx.db.insert("customers", {
          customerId: cid,
          name: trimmed,
          nameNormalized: n,
          status: "active",
          createdAt: Date.now(),
          updatedAt: Date.now(),
        });
      }
    }
    throw new Error("Failed to generate unique customerId");
  },
});

export const listActive = query({

function token(len = 28) {
  const chars = "abcdefghijklmnopqrstuvwxyz0123456789";
  let out = "";
  for (let i = 0; i < len; i++) out += chars[Math.floor(Math.random() * chars.length)];
  return out;
}

  args: {
    projectId: v.id("projects"),
    scope: v.union(v.literal("projectSummary"), v.literal("quote"), v.literal("gallery")),

import { action } from "./_generated/server";
import { v } from "convex/values";
import { normalizeName, newBusinessId } from "./lib/normalize";

export const migrateProjectsClientNameToCustomers = action({
  args: { dryRun: v.optional(v.boolean()) },
  handler: async (ctx, { dryRun }) => {
    const dry = !!dryRun;
    const projects = await ctx.db.query("projects").take(50_000);

    let touched = 0;

    for (const p of projects) {
      const hasCustomer = (p as any).customerId || (p as any).customerName;
      const clientName = (p as any).clientName?.trim();
      if (hasCustomer || !clientName) continue;

      const n = normalizeName(clientName);
      let customer = await ctx.db
        .query("customers")
        .withIndex("by_nameNormalized", (q) => q.eq("nameNormalized", n))
        .first();

      let customerId = customer?._id;
      if (!customerId && !dry) {
        // create customer
        let businessId = "";
        for (let i = 0; i < 5; i++) {
          const cid = newBusinessId("CUST");
          const clash = await ctx.db
            .query("customers")
            .withIndex("by_customerId", (q) => q.eq("customerId", cid))
            .first();
          if (!clash) { businessId = cid; break; }
        }
        if (!businessId) throw new Error("Failed to generate customerId");

        customerId = await ctx.db.insert("customers", {
          customerId: businessId,
          name: clientName,
          nameNormalized: n,
          status: "active",
          createdAt: Date.now(),
          updatedAt: Date.now(),
        });
      }

      if (!dry && customerId) {
        await ctx.db.patch(p._id, {
          customerId,
          customerName: clientName,
          updatedAt: Date.now(),
        });
      }

      touched++;
    }

    return { dryRun: dry, projectsProcessed: projects.length, projectsUpdated: touched };
  },
});

export const migrateTasksAssigneeToEmployeeIds = action({
  args: { dryRun: v.optional(v.boolean()) },
  handler: async (ctx, { dryRun }) => {
    const dry = !!dryRun;

    const employees = await ctx.db.query("employees").take(10_000);
    const map = new Map<string, string>();
    for (const e of employees) {
      map.set(normalizeName(e.displayName), e._id);
    }

    const tasks = await ctx.db.query("tasks").take(50_000);
    let updated = 0;

    for (const t of tasks) {
      const assignee = (t as any).assignee?.trim();
      const hasIds = Array.isArray((t as any).assigneeIds) && (t as any).assigneeIds.length > 0;
      if (!assignee || hasIds) continue;

      const id = map.get(normalizeName(assignee));
      if (!id) continue;

      if (!dry) {
        await ctx.db.patch(t._id, { assigneeIds: [id], updatedAt: Date.now() });
      }
      updated++;
    }

    return { dryRun: dry, tasksProcessed: tasks.length, tasksUpdated: updated };
  },
});

export const backfillQuoteVersionsCustomerFromProject = action({
  args: { dryRun: v.optional(v.boolean()) },
  handler: async (ctx, { dryRun }) => {
    const dry = !!dryRun;

import { query } from "./_generated/server";

export const health = query({
  args: {},
  handler: async (ctx) => {
    const customers = await ctx.db.query("customers").take(10_000);
    const ids = new Set<string>();
    const dupCustomerIds: string[] = [];

    for (const c of customers) {
      const cid = (c as any).customerId;
      if (ids.has(cid)) dupCustomerIds.push(cid);
      ids.add(cid);
    }

    const projects = await ctx.db.query("projects").take(10_000);
    const projectsWithCustomer = projects.filter((p) => (p as any).customerId).length;

    const tasks = await ctx.db.query("tasks").take(10_000);
    const tasksWithAssigneeText = tasks.filter((t) => (t as any).assignee).length;
    const tasksWithAssigneeIds = tasks.filter((t) => Array.isArray((t as any).assigneeIds) && (t as any).assigneeIds.length).length;

    return {
      customersCount: customers.length,
      dupCustomerIds,
      projectsCount: projects.length,
      projectsWithCustomer,
      tasksCount: tasks.length,
      tasksWithAssigneeText,
      tasksWithAssigneeIds,
    };
  },
});
