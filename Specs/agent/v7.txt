## File 3 — `changeSets_normalization_patch.ts`  
This is the “make it impossible to break again” fix. It normalizes **חומרים/עבודה** etc into canonical codes *before routing*.

```ts
// changeSets_normalization_patch.ts
// Drop these helpers into your changeSets applier before switch(op)

const HEBREW_TO_LINE_TYPE: Record<string, "material" | "work"> = {
  "חומרים": "material",
  "חומר": "material",
  "חומרי גלם": "material",
  "עבודה": "work",
  "עבודות": "work",
  "כוח אדם": "work",
};

function normalizeLineType(input: unknown): "material" | "work" {
  if (!input) throw new Error("Missing lineType");
  const s = String(input).trim();

  if (s === "material" || s === "materials") return "material";
  if (s === "work" || s === "labor") return "work";

  if (HEBREW_TO_LINE_TYPE[s]) return HEBREW_TO_LINE_TYPE[s];

  // last resort: contains match
  if (s.includes("חומר")) return "material";
  if (s.includes("עבוד")) return "work";

  throw new Error(`Unknown lineType value: ${s}`);
}

const HEBREW_SECTION_FALLBACKS: Record<string, string> = {
  "חומרים": "materials",
  "עבודה": "labor_direct",
  "הובלה": "transport",
  "התקנה": "install",
  "פירוק": "teardown_returns",
  "השכרות": "rental",
  "הדפסות": "printing_graphics",
  "גרפיקה": "printing_graphics",
  "ניהול": "management_overhead",
  "אוכל": "meals",
};

function normalizeSectionKey(sectionKey: unknown, sectionLabelHe?: unknown): string {
  if (sectionKey) return String(sectionKey).trim();
  const he = sectionLabelHe ? String(sectionLabelHe).trim() : "";
  return HEBREW_SECTION_FALLBACKS[he] || "misc";
}

// Optional guard: never allow Hebrew keys in op.data
function assertAsciiKeys(obj: any, path = "data") {
  if (!obj || typeof obj !== "object") return;
  for (const k of Object.keys(obj)) {
    // ASCII printable check
    if (!/^[\x20-\x7E]+$/.test(k)) {
      throw new Error(`Non-ASCII key detected at ${path}.${k} (Hebrew keys will break Convex payloads)`);
    }
    assertAsciiKeys(obj[k], `${path}.${k}`);
  }
}

/**
 * Pseudocode inside your applier:
 *
 * case "accountingLine.create":
 *   assertAsciiKeys(op.data)
 *   const lineType = normalizeLineType(op.data.lineType ?? op.data.type ?? op.data.accountingLineType)
 *   const sectionKey = normalizeSectionKey(op.data.sectionKey, op.data.sectionLabelHe)
 *   const sectionId = await resolveOrCreateSection(projectId, sectionKey, op.data.sectionLabelHe)
 *   const taskId = await resolveTaskRefIfNeeded(op.data.taskId, op.data.taskRef)
 *   route insert:
 *     if you have materialLines/workLines tables: insert into correct one
 *     else insert into unified accountingLines with lineType field
 */
````

This single shim fixes your “everything goes to other / cannot route” issue even if the model slips and outputs Hebrew type values.
